\fhsection{01}{The problem}{In Luxembourg, the advice on which provider to pick is often paid for by the provider.}

\begin{frame}{Fee-paid referrals}
  \framesubtitle{The problem}
  \fhdiagram[0.6]{diagrams/referral-problem}
\end{frame}

\begin{frame}{Why it hurts everyone except the middleman}
  \framesubtitle{The problem}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \fhcard{Fund managers overpay}{Proposals are bespoke, with hidden line items and out-of-pocket charges. Without a benchmark, nobody notices that the fee includes someone else's commission.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \fhcard{Advice isn't neutral}{The adviser recommending an administrator, depositary or ManCo may be paid by it, through referral fees, retrocessions or ``you send me, I send you'' arrangements.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \fhcard{Good providers are locked out}{Specialised, cost-effective firms can't win mandates on merit if introductions depend on who pays the introducer.}
    \end{column}
  \end{columns}
  \vspace{3mm}
  \begin{center}
    \fhcarddark{Our answer}{An independent comparator anyone can use, and advisors who sit at the table on the client's side, paid only by the client.}
  \end{center}
\end{frame}
