\fhsection{01}{The problem}{In Luxembourg, the advice on which provider to pick is often paid for by the provider.}

\begin{frame}{Fee-paid referrals: the hidden cost in fund services}
  \framesubtitle{The problem}
  \fhdiagram{diagrams/referral-problem}
\end{frame}

\begin{frame}{Why it hurts everyone except the middleman}
  \framesubtitle{The problem}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \fhcard{Fund managers overpay}{Proposals are bespoke, with hidden line items and out-of-pocket charges. Without a benchmark, nobody notices that the fee includes someone else's commission.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \fhcard{Advice isn't neutral}{The adviser recommending an administrator, depositary or ManCo may be paid by it, through referral fees, retrocessions or ``you send me, I send you'' arrangements.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \fhcard{Good providers are locked out}{Specialised, cost-effective firms can't win mandates on merit if introductions depend on who pays the introducer.}
    \end{column}
  \end{columns}
  \vspace{6mm}
  \begin{center}
    \fhcarddark{Our answer}{An independent comparator anyone can use, and advisors who sit at the table on the client's side, paid only by the client.}
  \end{center}
\end{frame}

\begin{frame}{Verified reviews: the comments system on providers}
  \framesubtitle{Trust}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{enumerate}\small
        \item A client writes a review on the provider's page: overall rating plus service quality, responsiveness, pricing transparency and technology.
        \item The reviewer gives a work email, company, role, relationship and client-since date; FundHub verifies the email.
        \item Automatic checks flag risks (free email domains, provider's own staff, duplicates).
        \item Staff verify the client relationship in the Console, then publish or reject.
        \item The review is published without the reviewer's identity and feeds the Peer Opinion Score and the FundHub Score.
      \end{enumerate}
    \end{column}
    \begin{column}{0.41\textwidth}
      \fhcard{Rules providers can't buy their way around}{\begin{itemize}\small
        \item Reviews can't be removed or edited on request
        \item Provider staff can't review themselves
        \item Subscriptions have no effect on reviews
      \end{itemize}}
      \vspace{3mm}
      \fhcard{Why it matters}{Reviews are the quality half of the score, and the aggregate feeds Data Intelligence: service levels by category and provider.}
    \end{column}
  \end{columns}
\end{frame}
